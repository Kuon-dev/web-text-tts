# Widgets Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Floating, draggable widget panels over the reader — a registry-driven framework plus its first two widgets, a clock and a pomodoro timer that pauses the voice on breaks.

**Architecture:** A widget is a pure `WidgetMeta` object (id, size, optional store factory, optional actions) plus a React face, registered in one frozen table that the settings page, the ⌘K palette and the floating layer all derive from. Widget state lives in small `subscribe`/`getSnapshot` stores outside React (the `player.ts` / `hud.ts` pattern); widgets reach the app only through narrow host ports (`playback`, `storage`, `notify`, `ticker`, `now`). Panel dragging is a hand-rolled pointer-capture hook that writes `transform` straight to the DOM and commits once on pointer-up; the reader tree is never touched.

**Tech Stack:** React 19, TypeScript 6, Vite 8, Tailwind 4, shadcn/ui (`radix-ui`), `motion` (LazyMotion `domAnimation`, `strict`), `lucide-react`, `sonner`, vitest 4 (node environment, `src/**/*.test.ts` only), oxlint.

**Spec:** `docs/superpowers/specs/2026-09-12-widgets-design.md`

## Global Constraints

- Every command below runs from the repo root (`web-text-tts/`) with `export PATH=~/node22/bin:$PATH` first.
- Tests: `npm test -w frontend -- <path>` runs one file; `npm test -w frontend` runs the suite. The environment is **node** — no DOM, no React rendering. Only `src/**/*.test.ts` is collected, so every module worth asserting is a pure `.ts` module.
- Type check + lint: `npm run build -w frontend` (runs `tsc -b`) and `npm run lint -w frontend`. Both must pass before the final ship commit.
- Path alias: `@/` = `frontend/src/`.
- Commit messages: `feat(widgets): …`, `test(widgets): …`, `build: ship …`, matching `git log`. Each task ends in a commit.
- No new dependencies. No `zustand`, no motion `domMax`.
- `Reader.tsx` is not modified by any task.
- Storage keys: layout `novel-tts:widgets`; per-widget `novel-tts:widget:<id>`.
- Keys claimed: `t` (start / pause pomodoro), `w` (show / hide widgets). `z` stays unbound (pinned by `keymap.test.ts`).

```ts
// frontend/src/lib/widgets/store.ts

/** The shape `player.ts` and `hud.ts` hand to useSyncExternalStore, with the
 *  listener bookkeeping written once. Widget stores wrap one of these. */
export interface ExternalStore<T> {
  subscribe(fn: () => void): () => void
  getSnapshot(): T
  set(next: T | ((prev: T) => T)): void
}

export function createExternalStore<T>(initial: T): ExternalStore<T> {
  let snap = initial
  const listeners = new Set<() => void>()
  return {
    subscribe: (fn) => {
      listeners.add(fn)
      return () => {
        listeners.delete(fn)
      }
    },
    getSnapshot: () => snap,
    // Same reference means nothing changed; useSyncExternalStore would
    // re-render every subscriber for a no-op otherwise.
    set: (next) => {
      const value = typeof next === "function" ? (next as (prev: T) => T)(snap) : next
      if (Object.is(value, snap)) return
      snap = value
      listeners.forEach((fn) => fn())
    },
  }
}
```

```ts
// frontend/src/lib/ticker.ts
import { useEffect, useState } from "react"

export type Cadence = "second" | "minute"

export interface Ticker {
  /** Call `fn(now)` on every boundary of `cadence` until the returned
   *  function is called. */
  every(cadence: Cadence, fn: (now: number) => void): () => void
}

const PERIOD: Record<Cadence, number> = { second: 1_000, minute: 60_000 }

/** Fires just after the boundary, never before it: a clock that ticks 50ms
 *  late shows the right minute; one that ticks 50ms early shows the old one
 *  for a whole minute. The dock clock used this same figure. */
const SLACK_MS = 50

interface Slot {
  listeners: Set<(now: number) => void>
  timer: ReturnType<typeof setTimeout> | undefined
}

/**
 * One timer per cadence, however many listeners. Listeners get `now` and
 * compute from it — the pomodoro stores a deadline and never counts ticks —
 * so a tab Chrome has throttled to one timer a minute still lands on the
 * right value when its late tick finally arrives.
 */
class WallClockTicker implements Ticker {
  private slots: Record<Cadence, Slot> = {
    second: { listeners: new Set(), timer: undefined },
    minute: { listeners: new Set(), timer: undefined },
  }

  constructor() {
    // A tab coming back to the foreground fires at once so the faces catch
    // up instead of waiting out the remainder of a throttled period.
    // Guarded like isMac(): vitest runs this module under node.
    if (typeof document !== "undefined") {
      document.addEventListener("visibilitychange", () => {
        if (document.visibilityState === "visible") {
          for (const c of Object.keys(this.slots) as Cadence[]) if (this.slots[c].listeners.size) this.fire(c)
        }
      })
    }
  }

  every(cadence: Cadence, fn: (now: number) => void): () => void {
    const slot = this.slots[cadence]
    slot.listeners.add(fn)
    if (slot.listeners.size === 1) this.schedule(cadence)
    return () => {
      slot.listeners.delete(fn)
      if (slot.listeners.size === 0) {
        clearTimeout(slot.timer)
        slot.timer = undefined
      }
    }
  }

  private schedule(cadence: Cadence) {
    const slot = this.slots[cadence]
    const period = PERIOD[cadence]
    clearTimeout(slot.timer)
    slot.timer = setTimeout(() => this.fire(cadence), period - (Date.now() % period) + SLACK_MS)
  }

  private fire(cadence: Cadence) {
    const slot = this.slots[cadence]
    const now = Date.now()
    slot.listeners.forEach((fn) => fn(now))
    if (slot.listeners.size) this.schedule(cadence)
  }
}

export const ticker: Ticker = new WallClockTicker()

/** The current time, re-rendering on each boundary of `cadence`. */
export function useNow(cadence: Cadence): number {
  const [now, setNow] = useState(() => Date.now())
  useEffect(() => ticker.every(cadence, setNow), [cadence])
  return now
}
```

```ts
// frontend/src/lib/widgets/types.ts
import type { ComponentType } from "react"
import type { LucideIcon } from "lucide-react"
import type { KeySpec } from "@/lib/keymap"
import type { Ticker } from "@/lib/ticker"

export interface Disposable {
  dispose(): void
}

/** A palette / key row a widget contributes. `id` is composed as
 *  `${widget.id}-${id}` — a meta declares `"toggle"`, the table carries
 *  `"pomodoro-toggle"`. `keys: []` means palette only. */
export interface WidgetAction<S> {
  id: string
  label: string
  keys: KeySpec[]
  run: (store: S) => void
}

/** Panel size in rem, so panels scale with the chrome under uiScale. */
export interface PanelSize {
  w: number
  h: number
}

/**
 * The pure half of a widget: everything the keymap, the settings page and the
 * node tests need, and nothing that imports React. A widget with neither a
 * store nor actions (the clock) is complete, not stubbed.
 */
export interface WidgetMeta<S = void> {
  /** Stable: it is the storage key, the palette id and the layout key. */
  id: string
  title: string
  size: PanelSize
  /** Which reader gutter a fresh panel lands in. */
  side: "left" | "right"
  /** Created when the widget is enabled, disposed when it is disabled. */
  createStore?: (host: WidgetHost) => S & Disposable
  actions?: readonly WidgetAction<S>[]
}

// The manifest is heterogeneous; `any` is the honest element type of a list
// of metas whose store types differ (`unknown` and `never` both fail one
// side of the variance).
export type AnyWidgetMeta = WidgetMeta<any>

/** The meta plus its face. `Component` may be `lazy()`; the frame renders
 *  it inside Suspense. */
export type WidgetDef<S = void> = WidgetMeta<S> & {
  Icon: LucideIcon
  Component: ComponentType<{ store: S }>
}
export type AnyWidgetDef = WidgetDef<any>

export interface PlaybackPort {
  isPlaying(): boolean
  pause(): void
  play(): void
}

/** Pre-scoped to the widget: the key is "novel-tts:widget:<id>". */
export interface StoragePort {
  /** `null` when absent, unparseable, or `parse` rejects it. */
  read<T>(parse: (raw: unknown) => T | null): T | null
  write(value: unknown): void
  clear(): void
}

export interface NotifyPort {
  toast(message: string, opts?: { id?: string; action?: { label: string; onClick: () => void } }): void
}

/** Everything a widget may touch outside itself. `host.ts` wires the real
 *  one; tests hand in a fake. */
export interface WidgetHost {
  playback: PlaybackPort
  storage: StoragePort
  notify: NotifyPort
  ticker: Ticker
  now(): number
}

/** px, viewport coordinates — a panel is `fixed` and translated by this. */
export interface PanelGeom {
  x: number
  y: number
}

export interface LayoutState {
  /** Enabled widgets in z-order; last is on top. */
  order: readonly string[]
  geom: Readonly<Record<string, PanelGeom>>
  hidden: boolean
}

/** What placement and clamping need to know about the screen. */
export interface LayoutEnv {
  vw: number
  vh: number
  /** The reader column's width in px (already capped to the viewport). */
  readerW: number
  /** The dock's height in px, published by Dock.tsx as --dock-h. */
  dockH: number
  /** Root font-size in px — uiScale changes it. */
  rem: number
}
```

```ts
// frontend/src/lib/widgets/layout.ts
import type { AnyWidgetMeta, LayoutEnv, LayoutState, PanelGeom, PanelSize } from "./types"

export const EMPTY_LAYOUT: LayoutState = { order: [], geom: {}, hidden: false }

/** The panel's title bar: the drag handle, and the part clampAll keeps on
 *  screen so a panel can always be grabbed back. Matches `h-9` in PanelFrame. */
export const TITLE_BAR_REM = 2.25
/** Between a panel and the reader column, and between stacked panels. */
export const GAP_REM = 1

export function enable(s: LayoutState, id: string, geom: PanelGeom): LayoutState {
  if (s.order.includes(id)) return s
  return { ...s, order: [...s.order, id], geom: { ...s.geom, [id]: geom } }
}

export function disable(s: LayoutState, id: string): LayoutState {
  if (!s.order.includes(id)) return s
  const geom = { ...s.geom }
  delete geom[id]
  return { ...s, order: s.order.filter((x) => x !== id), geom }
}

/** To the top of the stack. Z-order is DOM order, so this is the whole of it. */
export function raise(s: LayoutState, id: string): LayoutState {
  const i = s.order.indexOf(id)
  if (i < 0 || i === s.order.length - 1) return s
  return { ...s, order: [...s.order.filter((x) => x !== id), id] }
}

export function move(s: LayoutState, id: string, geom: PanelGeom): LayoutState {
  if (!s.order.includes(id)) return s
  return { ...s, geom: { ...s.geom, [id]: geom } }
}

export function toggleHidden(s: LayoutState): LayoutState {
  return { ...s, hidden: !s.hidden }
}

/** The reader column's left edge: the column is centred and capped to the
 *  viewport (`mx-auto` + `maxWidth: min(100%, …)` in Reader.tsx). */
const columnLeft = (env: LayoutEnv) => (env.vw - Math.min(env.vw, env.readerW)) / 2

/**
 * Where a freshly enabled panel goes: its side's gutter, one gap outside the
 * column, one gap below the lowest panel already on that side (or one gap
 * from the top). A gutter narrower than the panel still gets it — floating
 * panels may overlap the text, and the user drags it from there.
 */
export function placeNew(
  meta: Pick<AnyWidgetMeta, "size" | "side">,
  s: LayoutState,
  sizes: Readonly<Record<string, PanelSize>>,
  env: LayoutEnv,
): PanelGeom {
  const gap = GAP_REM * env.rem
  const left = columnLeft(env)
  const right = left + Math.min(env.vw, env.readerW)
  const x = meta.side === "right" ? right + gap : left - gap - meta.size.w * env.rem
  let y = gap
  for (const id of s.order) {
    const g = s.geom[id]
    const size = sizes[id]
    if (!g || !size) continue
    const onSide = meta.side === "right" ? g.x >= env.vw / 2 : g.x < env.vw / 2
    if (onSide) y = Math.max(y, g.y + size.h * env.rem + gap)
  }
  return { x, y }
}

const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v))

/**
 * After a resize (or a drag that ends off-screen): at least a title bar's
 * width of every panel stays inside the viewport horizontally, and the whole
 * title bar stays between the top edge and the dock vertically. Untouched
 * panels keep their geometry object, so memoised frames do not re-render.
 */
export function clampAll(s: LayoutState, sizes: Readonly<Record<string, PanelSize>>, env: LayoutEnv): LayoutState {
  const bar = TITLE_BAR_REM * env.rem
  let geom = s.geom
  for (const id of s.order) {
    const g = s.geom[id]
    const size = sizes[id]
    if (!g || !size) continue
    const w = size.w * env.rem
    const x = clamp(g.x, -(w - bar), env.vw - bar)
    const y = clamp(g.y, 0, Math.max(0, env.vh - env.dockH - bar))
    if (x !== g.x || y !== g.y) geom = { ...geom, [id]: { x, y } }
  }
  return geom === s.geom ? s : { ...s, geom }
}

const isGeom = (g: unknown): g is PanelGeom =>
  !!g && typeof g === "object" && Number.isFinite((g as PanelGeom).x) && Number.isFinite((g as PanelGeom).y)

/**
 * Validate-on-load, like reading.ts. Unknown ids go; an id without finite
 * geometry goes too — the widget is simply off until it is enabled again,
 * when placeNew positions it. Garbage yields the empty layout.
 */
export function parseLayout(raw: unknown, knownIds: readonly string[]): LayoutState {
  if (!raw || typeof raw !== "object") return EMPTY_LAYOUT
  const r = raw as { order?: unknown; geom?: unknown; hidden?: unknown }
  if (!Array.isArray(r.order)) return EMPTY_LAYOUT
  const geomIn = r.geom && typeof r.geom === "object" ? (r.geom as Record<string, unknown>) : {}
  const order: string[] = []
  const geom: Record<string, PanelGeom> = {}
  for (const id of r.order) {
    if (typeof id !== "string" || !knownIds.includes(id) || order.includes(id)) continue
    const g = geomIn[id]
    if (!isGeom(g)) continue
    order.push(id)
    geom[id] = { x: g.x, y: g.y }
  }
  if (!order.length && r.hidden !== true) return EMPTY_LAYOUT
  return { order, geom, hidden: r.hidden === true }
}
```

```ts
// frontend/src/widgets/pomodoro/machine.ts

/**
 * The pomodoro as pure functions of (state, config, now). Nothing here owns a
 * clock: a running session is a deadline (`endsAt`), and every question about
 * it is answered from the `now` passed in — so a tick that Chrome delivered a
 * minute late in a hidden tab still lands on the right answer.
 */

export type Phase = "work" | "short" | "long"

export interface Config {
  workMin: number
  shortMin: number
  longMin: number
  /** Work phases per cycle; the last one is followed by the long break. */
  roundsPerLong: number
  /** Pause the voice when a break starts. */
  pauseOnBreak: boolean
}

export const DEFAULT_CONFIG: Config = { workMin: 25, shortMin: 5, longMin: 15, roundsPerLong: 4, pauseOnBreak: true }

export const PHASE_LABEL: Record<Phase, string> = { work: "Work", short: "Short break", long: "Long break" }

/** `round` counts work phases completed in the current cycle. */
export type State =
  | { status: "idle"; phase: Phase; round: number }
  | { status: "running"; phase: Phase; round: number; endsAt: number }
  | { status: "paused"; phase: Phase; round: number; remainingMs: number }

export const IDLE: State = { status: "idle", phase: "work", round: 0 }

const MINUTE = 60_000
const PHASES: readonly Phase[] = ["work", "short", "long"]

const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v))
const num = (v: unknown, fallback: number, lo: number, hi: number) =>
  typeof v === "number" && Number.isFinite(v) ? clamp(v, lo, hi) : fallback

/** Every field validated on its own, so one bad value costs one default. */
export function clampConfig(raw: Partial<Config> | undefined): Config {
  const d = DEFAULT_CONFIG
  return {
    workMin: num(raw?.workMin, d.workMin, 1, 120),
    shortMin: num(raw?.shortMin, d.shortMin, 1, 120),
    longMin: num(raw?.longMin, d.longMin, 1, 120),
    roundsPerLong: Math.round(num(raw?.roundsPerLong, d.roundsPerLong, 2, 8)),
    pauseOnBreak: typeof raw?.pauseOnBreak === "boolean" ? raw.pauseOnBreak : d.pauseOnBreak,
  }
}

export function phaseMs(phase: Phase, cfg: Config): number {
  return (phase === "work" ? cfg.workMin : phase === "short" ? cfg.shortMin : cfg.longMin) * MINUTE
}

export function start(s: State, cfg: Config, now: number): State {
  if (s.status !== "idle") return s
  return { status: "running", phase: s.phase, round: s.round, endsAt: now + phaseMs(s.phase, cfg) }
}

export function pause(s: State, now: number): State {
  if (s.status !== "running") return s
  return { status: "paused", phase: s.phase, round: s.round, remainingMs: Math.max(0, s.endsAt - now) }
}

export function resume(s: State, now: number): State {
  if (s.status !== "paused") return s
  return { status: "running", phase: s.phase, round: s.round, endsAt: now + s.remainingMs }
}

export function remaining(s: State, cfg: Config, now: number): number {
  switch (s.status) {
    case "idle":
      return phaseMs(s.phase, cfg)
    case "running":
      return Math.max(0, s.endsAt - now)
    case "paused":
      return s.remainingMs
  }
}

export function nextPhase(phase: Phase, round: number, cfg: Config): { phase: Phase; round: number } {
  if (phase === "work") {
    const done = round + 1
    return done >= cfg.roundsPerLong ? { phase: "long", round: done } : { phase: "short", round: done }
  }
  return { phase: "work", round: phase === "long" ? 0 : round }
}

/** Crosses the deadline if it has passed. The next phase is anchored to `now`,
 *  not to the old deadline: a tick that arrives ten minutes late must not
 *  produce a phase that is already over, or the ticks after it would chain
 *  through phases and toasts nobody saw. */
export function advance(s: State, cfg: Config, now: number): { state: State; entered: Phase | null } {
  if (s.status !== "running" || now < s.endsAt) return { state: s, entered: null }
  const next = nextPhase(s.phase, s.round, cfg)
  return { state: { status: "running", ...next, endsAt: now + phaseMs(next.phase, cfg) }, entered: next.phase }
}

/** Ends the phase now. A skipped work phase is not a completed round — the
 *  break that follows is whichever the cycle is due, but `round` stays. A
 *  running session keeps running into the next phase; anything else lands
 *  idle there. */
export function skip(s: State, cfg: Config, now: number): { state: State; entered: Phase } {
  const next =
    s.phase === "work"
      ? { phase: (s.round + 1 >= cfg.roundsPerLong ? "long" : "short") as Phase, round: s.round }
      : nextPhase(s.phase, s.round, cfg)
  const state: State =
    s.status === "running" ? { status: "running", ...next, endsAt: now + phaseMs(next.phase, cfg) } : { status: "idle", ...next }
  return { state, entered: next.phase }
}

/** "mm:ss". Ceiling, so a fresh phase reads its full length and the display
 *  reaches 00:00 on the deadline rather than a second early. */
export function formatRemaining(ms: number): string {
  const total = Math.ceil(Math.max(0, ms) / 1000)
  const m = Math.floor(total / 60)
  const s = total % 60
  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`
}

/**
 * A saved session, validated. A running session whose deadline passed while
 * the app was closed comes back **paused at the start of the next phase** —
 * one step, not a replay of every phase since: the user was not there for the
 * bell, so the store makes no toast and no playback change on restore.
 */
export function parseSession(raw: unknown, now: number): { config: Config; state: State } | null {
  if (!raw || typeof raw !== "object") return null
  const r = raw as { config?: unknown; state?: unknown }
  const config = clampConfig(r.config && typeof r.config === "object" ? (r.config as Partial<Config>) : undefined)
  const s = r.state as { status?: unknown; phase?: unknown; round?: unknown; endsAt?: unknown; remainingMs?: unknown } | undefined
  if (!s || typeof s !== "object") return null
  if (!PHASES.includes(s.phase as Phase)) return null
  if (typeof s.round !== "number" || !Number.isInteger(s.round) || s.round < 0) return null
  const base = { phase: s.phase as Phase, round: Math.min(s.round, config.roundsPerLong) }
  switch (s.status) {
    case "idle":
      return { config, state: { status: "idle", ...base } }
    case "paused":
      if (typeof s.remainingMs !== "number" || !Number.isFinite(s.remainingMs) || s.remainingMs < 0) return null
      return { config, state: { status: "paused", ...base, remainingMs: s.remainingMs } }
    case "running": {
      if (typeof s.endsAt !== "number" || !Number.isFinite(s.endsAt)) return null
      if (s.endsAt > now) return { config, state: { status: "running", ...base, endsAt: s.endsAt } }
      const next = nextPhase(base.phase, base.round, config)
      return { config, state: { status: "paused", ...next, remainingMs: phaseMs(next.phase, config) } }
    }
    default:
      return null
  }
}
```

```ts
// frontend/src/widgets/pomodoro/store.ts
import { createExternalStore } from "@/lib/widgets/store"
import type { Disposable, WidgetHost } from "@/lib/widgets/types"
import {
  DEFAULT_CONFIG,
  IDLE,
  advance,
  clampConfig,
  formatRemaining,
  parseSession,
  pause,
  phaseMs,
  resume,
  skip,
  start,
  type Config,
  type Phase,
  type State,
} from "./machine"

export interface PomodoroSnapshot {
  config: Config
  state: State
  /** Bumped once a second while running, so the face re-renders and reads
   *  the countdown off the deadline. Nothing else changes per tick. */
  tick: number
}

export interface PomodoroStore extends Disposable {
  subscribe(fn: () => void): () => void
  getSnapshot(): PomodoroSnapshot
  start(): void
  pause(): void
  resume(): void
  toggle(): void
  skip(): void
  reset(): void
  setConfig(patch: Partial<Config>): void
}

/** One id, so a phase's toast replaces the previous one instead of stacking. */
const TOAST_ID = "pomodoro"
const ENTER_LABEL: Record<Phase, string> = { work: "Back to work", short: "Short break", long: "Long break" }

/**
 * The pomodoro's live state, outside React so the keymap can reach it and a
 * reload can restore it. Holds a ticker subscription only while running: an
 * idle or paused timer costs nothing.
 */
export function createPomodoroStore(host: WidgetHost): PomodoroStore {
  const saved = host.storage.read((raw) => parseSession(raw, host.now()))
  const store = createExternalStore<PomodoroSnapshot>({
    config: saved?.config ?? DEFAULT_CONFIG,
    state: saved?.state ?? IDLE,
    tick: 0,
  })
  let offTick: (() => void) | null = null

  const snap = () => store.getSnapshot()

  // Transitions and pauses only — a running session is a deadline, and the
  // deadline does not change per second.
  const persist = () => {
    const { config, state } = snap()
    host.storage.write({ config, state })
  }

  const announce = (entered: Phase) => {
    const { config } = snap()
    const length = formatRemaining(phaseMs(entered, config))
    if (entered === "work") {
      // Never auto-resume: the room may be empty. Offer it instead.
      host.notify.toast(`${ENTER_LABEL.work} — ${length}`, {
        id: TOAST_ID,
        action: { label: "Resume reading", onClick: () => host.playback.play() },
      })
      return
    }
    if (config.pauseOnBreak) host.playback.pause()
    host.notify.toast(`${ENTER_LABEL[entered]} — ${length}`, { id: TOAST_ID })
  }

  const onTick = (now: number) => {
    const s = snap()
    const { state, entered } = advance(s.state, s.config, now)
    store.set({ ...s, state, tick: s.tick + 1 })
    if (entered) {
      announce(entered)
      persist()
    }
  }

  const syncTicker = () => {
    const running = snap().state.status === "running"
    if (running && !offTick) offTick = host.ticker.every("second", onTick)
    if (!running && offTick) {
      offTick()
      offTick = null
    }
  }

  const commit = (state: State) => {
    const s = snap()
    if (state === s.state) return
    store.set({ ...s, state })
    persist()
    syncTicker()
  }

  // A restored running session ticks from the first second.
  syncTicker()

  return {
    subscribe: store.subscribe,
    getSnapshot: store.getSnapshot,
    start: () => commit(start(snap().state, snap().config, host.now())),
    pause: () => commit(pause(snap().state, host.now())),
    resume: () => commit(resume(snap().state, host.now())),
    toggle: () => {
      const { state, config } = snap()
      const now = host.now()
      commit(state.status === "running" ? pause(state, now) : state.status === "paused" ? resume(state, now) : start(state, config, now))
    },
    skip: () => {
      const { state, config } = snap()
      const r = skip(state, config, host.now())
      commit(r.state)
      announce(r.entered)
    },
    reset: () => commit(IDLE),
    setConfig: (patch) => {
      const s = snap()
      store.set({ ...s, config: clampConfig({ ...s.config, ...patch }) })
      persist()
    },
    dispose: () => {
      offTick?.()
      offTick = null
      host.storage.clear()
    },
  }
}
```

```ts
// frontend/src/lib/widgets/runtime.ts
import { WIDGET_MANIFEST } from "./manifest"
import type { AnyWidgetMeta, Disposable, LayoutState, WidgetHost } from "./types"

export interface LayoutSource {
  subscribe(fn: () => void): () => void
  getSnapshot(): LayoutState
}

/**
 * A widget's live state exists exactly while the widget is enabled: the
 * runtime follows the layout and creates a store when an id enters `order`,
 * disposes it when the id leaves. Keymap actions reach a store through
 * `get`; a disabled widget makes its action a no-op.
 *
 * The host factory is injected rather than imported so this module — which
 * keymap.ts pulls in — never touches player.ts, whose module scope constructs
 * an <audio> element that does not exist under node.
 */
export class WidgetRuntime {
  private stores = new Map<string, Disposable>()

  constructor(private readonly metas: readonly AnyWidgetMeta[]) {}

  /** Sync now and on every layout change. The returned function only stops
   *  following — it disposes nothing, so React's StrictMode effect re-run
   *  (mount, cleanup, mount) cannot clear a widget's saved session. */
  follow(layout: LayoutSource, createHost: (id: string) => WidgetHost): () => void {
    const sync = () => this.sync(layout.getSnapshot().order, createHost)
    sync()
    return layout.subscribe(sync)
  }

  get<S>(id: string): (S & Disposable) | undefined {
    return this.stores.get(id) as (S & Disposable) | undefined
  }

  private sync(order: readonly string[], createHost: (id: string) => WidgetHost) {
    for (const id of order) {
      if (this.stores.has(id)) continue
      const meta = this.metas.find((m) => m.id === id)
      if (meta?.createStore) this.stores.set(id, meta.createStore(createHost(id)))
    }
    for (const [id, store] of this.stores) {
      if (order.includes(id)) continue
      store.dispose()
      this.stores.delete(id)
    }
  }
}

export const runtime = new WidgetRuntime(WIDGET_MANIFEST)
```

```ts
// frontend/src/lib/widgets/layout-store.ts
import { useSyncExternalStore } from "react"
import { EMPTY_LAYOUT, clampAll, disable, enable, move, parseLayout, placeNew, raise, toggleHidden } from "./layout"
import { WIDGET_IDS, WIDGET_SIZES, metaById } from "./manifest"
import { createExternalStore } from "./store"
import type { LayoutEnv, LayoutState, PanelGeom } from "./types"

const STORAGE_KEY = "novel-tts:widgets"

/** What env() reports where there is no window: vitest under node. */
const FALLBACK = { vw: 1280, vh: 800, rem: 16, dockH: 0 }

function load(): LayoutState {
  try {
    const raw = localStorage.getItem(STORAGE_KEY)
    return raw ? parseLayout(JSON.parse(raw), WIDGET_IDS) : EMPTY_LAYOUT
  } catch {
    return EMPTY_LAYOUT // no localStorage under node; garbage in the browser
  }
}

function save(s: LayoutState) {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(s))
  } catch {
    /* private mode etc. — the layout just won't persist */
  }
}

/**
 * Where the panels are, which are enabled, and whether they are hidden —
 * device-scoped in localStorage, because positions are screen geometry.
 * Every mutation goes through the pure functions in layout.ts; this class
 * adds the screen (viewport, root font-size, the dock's published height,
 * the reader width App pushes in) and the persistence.
 */
export class LayoutStore {
  private store = createExternalStore<LayoutState>(load())
  private readerW = FALLBACK.vw

  subscribe = this.store.subscribe
  getSnapshot = this.store.getSnapshot

  constructor() {
    if (typeof window === "undefined") return
    // A shrink can strand a panel off-screen; one clamp per frame is plenty.
    let raf = 0
    window.addEventListener("resize", () => {
      cancelAnimationFrame(raf)
      raf = requestAnimationFrame(() => this.commit(clampAll(this.getSnapshot(), WIDGET_SIZES, this.env())))
    })
  }

  private env(): LayoutEnv {
    if (typeof window === "undefined") return { ...FALLBACK, readerW: this.readerW }
    const root = getComputedStyle(document.documentElement)
    return {
      vw: window.innerWidth,
      vh: window.innerHeight,
      rem: parseFloat(root.fontSize) || FALLBACK.rem,
      dockH: parseFloat(root.getPropertyValue("--dock-h")) || 0,
      readerW: this.readerW,
    }
  }

  private commit(next: LayoutState) {
    if (next === this.getSnapshot()) return
    this.store.set(next)
    save(next)
  }

  /** The reader column's max width in px — App keeps this current from
   *  prefs.width. It only shapes future placements, so it does not emit. */
  setReaderWidth(px: number) {
    this.readerW = px
  }

  enable(id: string) {
    const meta = metaById(id)
    const s = this.getSnapshot()
    if (!meta || s.order.includes(id)) return
    this.commit(enable(s, id, placeNew(meta, s, WIDGET_SIZES, this.env())))
  }

  disable(id: string) {
    this.commit(disable(this.getSnapshot(), id))
  }

  toggle(id: string) {
    if (this.getSnapshot().order.includes(id)) this.disable(id)
    else this.enable(id)
  }

  raise(id: string) {
    this.commit(raise(this.getSnapshot(), id))
  }

  /** Clamped: a drag that ends off-screen still leaves the title bar reachable. */
  move(id: string, geom: PanelGeom) {
    this.commit(clampAll(move(this.getSnapshot(), id, geom), WIDGET_SIZES, this.env()))
  }

  toggleHidden() {
    this.commit(toggleHidden(this.getSnapshot()))
  }

  /** Every widget off, positions forgotten. */
  reset() {
    this.commit(EMPTY_LAYOUT)
  }
}

export const layout = new LayoutStore()

export function useWidgetLayout(): LayoutState {
  return useSyncExternalStore(layout.subscribe, layout.getSnapshot)
}
```

```ts
      run: () => deps.layout.toggleHidden(),

    rows.push({
      id: `widget-${m.id}`,
      label: `Toggle ${m.title.toLowerCase()}`,
      group: "Widgets",
      scope: "reader",
      keys: [],
      run: () => deps.layout.toggle(m.id),
```

```ts
          return raw === null ? null : parse(JSON.parse(raw))

          return null // garbage, or no storage at all

        try {
          localStorage.setItem(key, JSON.stringify(value))
        } catch {
          /* private mode etc. — the session just won't survive a reload */
        }
      },
      clear: () => {
        try {
          localStorage.removeItem(key)
        } catch {
          /* same */
        }
      },
    },
    notify: {
      toast: (message, opts) => {
        toast(message, { id: opts?.id, action: opts?.action })
      },
    },
    ticker,
    now: () => Date.now(),
  }
```

```ts
// frontend/src/components/widgets/useDrag.ts
import { useCallback, type PointerEvent as ReactPointerEvent, type RefObject } from "react"
import { layout } from "@/lib/widgets/layout-store"

/**
 * Drag a panel by its title bar. Pointer capture on the handle, `transform`
 * written straight to the panel's outer element on every move, and a single
 * `layout.move` when the pointer lifts — React renders nothing during the
 * drag, and `transform` is compositor-only. Hand-rolled because App mounts
 * LazyMotion with `domAnimation` + `strict`, and motion's `drag` lives in the
 * heavier `domMax` bundle.
 */
export function useDrag(outer: RefObject<HTMLDivElement | null>, id: string) {
  return useCallback(
    (e: ReactPointerEvent<HTMLElement>) => {
      if (e.button !== 0) return // primary button only
      const el = outer.current
      const from = layout.getSnapshot().geom[id]
      if (!el || !from) return
      const handle = e.currentTarget
      const dx = e.clientX - from.x
      const dy = e.clientY - from.y
      let last = from

      const onMove = (ev: PointerEvent) => {
        last = { x: ev.clientX - dx, y: ev.clientY - dy }
        el.style.transform = `translate(${last.x}px, ${last.y}px)`
      }
      const onUp = () => {
        handle.removeEventListener("pointermove", onMove)
        handle.removeEventListener("pointerup", onUp)
        handle.removeEventListener("pointercancel", onUp)
        if (handle.hasPointerCapture(e.pointerId)) handle.releasePointerCapture(e.pointerId)
        layout.move(id, last) // clamped by the store; the frame's effect re-applies it
      }

      handle.setPointerCapture(e.pointerId)
      handle.addEventListener("pointermove", onMove)
      handle.addEventListener("pointerup", onUp)
      handle.addEventListener("pointercancel", onUp)
      e.preventDefault() // no text selection while dragging
    },
    [outer, id],
  )
}
```

```tsx
// frontend/src/components/widgets/PanelFrame.tsx
import { Suspense, memo, useLayoutEffect, useRef } from "react"
import { X } from "lucide-react"
import { m } from "motion/react"
import type { Transition } from "motion/react"
import { Button } from "@/components/ui/button"
import { layout } from "@/lib/widgets/layout-store"
import type { AnyWidgetDef, PanelGeom } from "@/lib/widgets/types"
import { useDrag } from "./useDrag"

const PANEL_SPRING: Transition = { type: "spring", stiffness: 300, damping: 28 }

interface Props {
  def: AnyWidgetDef
  geom: PanelGeom
  store: unknown
}

/**
 * The chrome around a widget: a title bar that is the drag handle, a close
 * button, and the enter/exit spring. Two nested elements by necessity — the
 * outer div is fixed and owns `translate`, the drag hook's property; the
 * inner m.div owns opacity/scale, motion's. The conflict AdjustHUD.tsx
 * documents, resolved the same way.
 *
 * The translate is applied in a layout effect, never through the `style`
 * prop: a re-render of the panel's content mid-drag (a pomodoro tick) must
 * not hand React a stale transform to write back under the pointer.
 *
 * Memoised: `def` and `store` are stable, and `geom` changes identity only
 * when this panel moved — a `raise` re-renders nothing but the order.
 */
export const PanelFrame = memo(function PanelFrame({ def, geom, store }: Props) {
  const outer = useRef<HTMLDivElement>(null)
  useLayoutEffect(() => {
    const el = outer.current
    if (el) el.style.transform = `translate(${geom.x}px, ${geom.y}px)`
  }, [geom.x, geom.y])
  const onHandlePointerDown = useDrag(outer, def.id)
  const { Icon, Component, title, size } = def

  return (
    <div
      ref={outer}
      role="region"
      aria-label={title}
      className="pointer-events-auto fixed top-0 left-0"
      style={{ width: `${size.w}rem`, height: `${size.h}rem` }}
      onPointerDownCapture={() => layout.raise(def.id)}
    >
      <m.div
        initial={{ opacity: 0, scale: 0.96 }}
        animate={{ opacity: 1, scale: 1 }}
        exit={{ opacity: 0, scale: 0.96 }}
        transition={PANEL_SPRING}
        className="flex h-full flex-col overflow-hidden rounded-lg border bg-card/85 shadow-lg backdrop-blur"
      >
        {/* h-9 is TITLE_BAR_REM (2.25rem) in layout.ts: the strip clampAll
            keeps on screen so a panel can always be grabbed back. */}
        <div
          className="flex h-9 shrink-0 cursor-grab touch-none items-center gap-1.5 border-b px-2 select-none active:cursor-grabbing"
          onPointerDown={onHandlePointerDown}
        >
          <Icon className="size-3.5 text-muted-foreground" aria-hidden />
          <span className="min-w-0 flex-1 truncate text-xs font-medium">{title}</span>
          <Button
            variant="ghost"
            size="icon-xs"
            className="text-muted-foreground"
            aria-label={`Close ${title}`}
            // Not a drag start — the bar's handler must not see this press.
            onPointerDown={(e) => e.stopPropagation()}
            onClick={() => layout.disable(def.id)}
          >
            <X aria-hidden />
          </Button>
        </div>
        <div className="min-h-0 flex-1">
          <Suspense fallback={null}>
            <Component store={store} />
          </Suspense>
        </div>
      </m.div>
    </div>
  )
})
```
